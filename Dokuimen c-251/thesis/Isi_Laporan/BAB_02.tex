    \subsection{Representasi Sistem dalam Domain Laplace dan Integrasi dengan Flowchart}
    \label{subsec:laplace_block_diagram}
    
    Pada sistem kontrol tertutup, sinyal-sinyal yang terlibat didefinisikan sebagai berikut:
    \begin{itemize}
        \item $R(s)$ : sinyal referensi (setpoint)
        \item $Y(s)$ : sinyal keluaran sistem
        \item $E(s)$ : sinyal error, dengan $E(s) = R(s) - Y(s)$
        \item $U(s)$ : sinyal kendali
        \item $G(s)$ : fungsi alih (\textit{transfer function}) plant
        \item $C(s)$ : fungsi alih pengendali (controller)
    \end{itemize}
    
    Hubungan antar variabel dalam domain Laplace mengikuti prinsip sistem kontrol umpan balik, yaitu:
    \begin{equation}
        E(s) = R(s) - Y(s)
    \end{equation}
    
    \begin{equation}
        U(s) = C(s)E(s)
    \end{equation}
    
    \begin{equation}
        Y(s) = G(s)U(s)
    \end{equation}
    
    Sehingga diperoleh fungsi alih sistem tertutup:
    \begin{equation}
        \frac{Y(s)}{R(s)} = \frac{C(s)G(s)}{1 + C(s)G(s)}
    \end{equation}
    
    \noindent Representasi grafis dari hubungan tersebut ditunjukkan pada Gambar~\ref{fig:block_diagram_control}.
    
    \begin{figure}[!ht]
    \centering
    \begin{tikzpicture}[auto, node distance=2.5cm, >=latex']
    
    % Nodes
    \node [coordinate] (input) {};
    \node [circle, draw, right of=input] (sum) {$\sum$};
    \node [rectangle, draw, right of=sum, minimum width=1.5cm] (controller) {$C(s)$};
    \node [rectangle, draw, right of=controller, minimum width=1.5cm] (plant) {$G(s)$};
    \node [coordinate, right of=plant] (output) {};
    
    % Connections
    \draw[->] (input) -- node {$R(s)$} (sum);
    \draw[->] (sum) -- node {$E(s)$} (controller);
    \draw[->] (controller) -- node {$U(s)$} (plant);
    \draw[->] (plant) -- node {$Y(s)$} (output);
    
    % Feedback
    \draw[->] (output) |- ++(0,-1.5) -| node[pos=0.25] {$-$} (sum);
    
    \end{tikzpicture}
    \caption{Block Diagram Sistem Kontrol Tertutup dalam Domain Laplace}
    \label{fig:block_diagram_control}
    \end{figure}
    
    \noindent Keterkaitan antara flowchart algoritme (Gambar~\ref{fig:flowchart_control}) dan block diagram pada Gambar~\ref{fig:block_diagram_control} dapat dijelaskan sebagai berikut:
    
    \begin{enumerate}
        \item Proses pembacaan setpoint dan output pada flowchart merepresentasikan sinyal $R(s)$ dan $Y(s)$.
        \item Tahap perhitungan error $e = r - y$ sesuai dengan operasi penjumlahan pada \textit{summing junction} yang menghasilkan $E(s)$.
        \item Proses perhitungan sinyal kendali pada flowchart merupakan implementasi dari fungsi pengendali $C(s)$.
        \item Tahap aktuasi sistem merepresentasikan penerapan sinyal $U(s)$ ke plant $G(s)$.
        \item Pengukuran output kembali ke sistem menunjukkan mekanisme umpan balik (feedback).
        \item Keputusan kondisi berhenti pada flowchart tidak direpresentasikan secara eksplisit dalam domain Laplace, namun berkaitan dengan analisis kestabilan dan respon sistem terhadap error.
    \end{enumerate}

    \subsection{Algoritme}
    \label{subsec:algo}
    
    Anda dapat menulis algortime seperti pada contoh "Algoritme~\ref{algo:algo1}". 
    \begin{algorithm}
    	\caption{PPO} 
    	\label{algo:algo1}
    	\begin{algorithmic}[1]
    		\For {$iteration=1,2,\ldots$}
    			\For {$actor=1,2,\ldots,N$}
    				\State Run policy $\pi_{\theta_{old}}$ in environment for $T$ time steps
    				\State Compute advantage estimates $\hat{A}_{1},\ldots,\hat{A}_{T}$
    			\EndFor
    			\State Optimize surrogate $L$ wrt. $\theta$, with $K$ epochs and minibatch size $M\leq NT$
    			\State $\theta_{old}\leftarrow\theta$
    		\EndFor
    	\end{algorithmic} 
    \end{algorithm}
